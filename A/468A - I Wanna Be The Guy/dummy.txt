#include<iostream>
using namespace std;
int main(){
    int n;
    cin>>n;
    int a[n];
    for(int i=0; i<n; i++)
        a[i]=i+1;
    int x,y;
    cin>>x;
    int xl[x];
    for(int i=0;i<x; i++)
        cin>>xl[i];
    cin>>y;
    int yl[y];
    for(int i=0;i<y; i++)
        cin>>yl[i];
    int done;
    for(int i=0;i<x;i++){
        for(int j=0;j<x;j++){
            if(i==j) continue;
            else{
                for(int k=0;k<n;k++){
                    if(xl[i]==a[n]) done++;
                }
            }
        }
    }
    for(int i=0;i<y;i++){
        for(int j=0;j<y;j++){
            if(i==j) continue;
            else{
                for(int k=0;k<n;k++){
                    if(yl[i]==a[n]) done++;
                }
            }
        }
    }
    if(done==n)
        cout<<"I become the guy.";
    else
        cout<<"Oh, my keyboard!";
    return 0;
}
